\section{Main Theorem}

\begin{theorem}[Order-optimal fully non-adaptive one-bit mean estimation]
\label{thm:main}
Under Assumptions~\ref{assump:parameter-domain}--
\ref{assump:precommitted-protocol}, for every fixed $k>1$ there are constants
$c_k,C_k>0$ and choices $a_k,b_k,\alpha_k,\beta_k>0$, all depending only on
$k$, such that the protocol of Section~\ref{sec:preliminaries} is well defined
for every known $\lambda\geq\sigma>0$, every
$0<\epsilon\leq c_k\sigma$, and every $\delta\in(0,1/2)$.  Every query is a
Borel set fixed before the first response; the protocol has no interactive
transition, sends exactly one bit for each independent sample it uses, and has
the deterministic non-stopping horizon $n=N_{\rm loc}+N_{\rm ref}$ satisfying
\[
 n\leq C_k r_k(\lambda,\sigma,\epsilon,\delta).
\]
Moreover, with probability taken jointly over all observations and all
localization and refinement randomness,
\[
 \sup_{D\in\mathcal D(k,\lambda,\sigma)}
 \Pr_D\left\{|\widehat\mu-\mu(D)|>\epsilon\right\}\leq\delta.
\]
The norm is absolute value on $\mathbb R$; the displayed rate retains every
$\lambda,\sigma,\epsilon,\delta$ factor, and the hidden constant depends only
on the fixed $k$.

For any realized localization output $c$, the same construction has two exact
baseline reductions.  If $D(J_{0,j_0(c)})=1$, all retained higher-level
corrections and the outer residual vanish exactly, leaving the same fixed
median aggregation applied only to the level-zero unbiased dither correction.
If $D\{m_0(c)\}=1$, every refinement correction vanishes seedwise and
$\widehat\mu=m_0(c)=\mu(D)$.
\end{theorem}

\begin{corollary}[Explicit three-regime rate specialization]
\label{cor:explicit-rate-specialization}
Let $C_k^{\rm rec}$ and $C_k^{\rm var}$ be the finite $k$-only constants
defined in Lemma~\ref{lem:recentered-moment} and Proposition~
\ref{prop:conditional-variance}.  For fixed $k>1$, choose
\[
 a_k\geq200,
 \quad b_k=\max\{a_k,(4C_k^{\rm rec})^{1/(k-1)}\},
 \quad \beta_k=16C_k^{\rm var},
 \quad \alpha_k=4,
\]
and
\[
 0<c_k\leq\min\left\{\frac12,
 \left(\frac{b_k}{2a_k}\right)^{k-1}\right\}.
\]
Then, for every known $\lambda\geq\sigma>0$, every
$0<\epsilon\leq c_k\sigma$, and every $\delta\in(0,1/2)$, the fully
non-adaptive protocol precommits all queries, sends one bit per independent
sample, and has deterministic non-stopping horizon
$n=N_{\rm loc}+N_{\rm ref}$ satisfying
\[
 n\leq C_k\left[\log\frac{\lambda}{\sigma}+
 \begin{cases}
 \dfrac{\sigma^2}{\epsilon^2}\log\dfrac1\delta,&k>2,\\[0.6em]
 \dfrac{\sigma^2}{\epsilon^2}\log\dfrac{\sigma}{\epsilon}
 \log\dfrac1\delta,&k=2,\\[0.6em]
 \left(\dfrac{\sigma}{\epsilon}\right)^{k/(k-1)}
 \log\dfrac1\delta,&1<k<2.
 \end{cases}\right]
\]
Moreover,
\[
 \sup_{D\in\mathcal D(k,\lambda,\sigma)}
 \Pr_D\left\{|\widehat\mu-\mu(D)|>\epsilon\right\}\leq\delta.
\]
The probability is unconditional and joint over all samples and protocol
randomness, the norm is absolute value on $\mathbb R$, and $C_k$ and every
hidden constant depend only on the fixed $k$.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:main} gives the fixed-horizon protocol and its unconditional
absolute-error guarantee.  Proposition~\ref{prop:rate-specialization-bridge}
verifies the displayed auxiliary choices, all ceiling and probability
conditions, and the stated three-regime bound with exactly the single
$\log(\sigma/\epsilon)$ factor when $k=2$.
\end{proof}
